% What happens between "a node finishes" and "the simulator runs a dependent", before and with --lazy-decoding.
\begin{tikzpicture}[
  stage/.style={box,text width=29mm,minimum height=13mm},
  cost/.style={font=\scriptsize,align=center,text width=31mm},
  bad/.style={cost,text=red!65!black},
  good/.style={cost,text=green!40!black},
  rowlabel/.style={font=\small\bfseries,anchor=east}]
\foreach \y/\label in {0/Before,-3.1/Lazy} \node[rowlabel] at (-1.9,\y) {\label};
% Before: counters live in the mapped file, ready nodes are decoded copies.
\node[stage,fill=black!5] (e1) at (0,0) {A node\\finishes};
\node[stage,fill=green!12] (e2) at (4.1,0) {Decrement counters\\\textbf{in the file}};
\node[stage,fill=green!12] (e3) at (8.2,0) {Counter 0:\\\textbf{decode} the node};
\node[stage,fill=orange!20] (e4) at (12.3,0) {Ready list:\\decoded copy};
\node[stage,fill=black!5] (e5) at (16.4,0) {Simulator\\runs it};
\node[bad,below=1mm of e2] {first write copies\\a 4\,KiB page};
\node[bad,below=1mm of e4] {80\,B per ready node};
% Lazy: counters in an array, ready nodes are indices decoded when taken.
\node[stage,fill=black!5] (l1) at (0,-3.1) {A node\\finishes};
\node[stage,fill=green!12] (l2) at (4.1,-3.1) {Decrement counters\\\textbf{in an array}};
\node[stage,fill=green!12] (l3) at (8.2,-3.1) {Counter 0:\\remember the index};
\node[stage,fill=orange!20] (l4) at (12.3,-3.1) {Ready list:\\index};
\node[stage,fill=black!5] (l5) at (16.4,-3.1) {Simulator \textbf{decodes}\\and runs it};
\node[good,below=1mm of l2] {4\,B per node;\\file never written};
\node[good,below=1mm of l4] {4\,B per ready node};
\node[good,below=1mm of l5] {decoded exactly once};
\foreach \r in {e,l} \foreach \a/\b in {1/2,2/3,3/4,4/5} \draw[arrow] (\r\a)--(\r\b);
\end{tikzpicture}
