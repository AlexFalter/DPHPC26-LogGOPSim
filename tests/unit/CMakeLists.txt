# Unit tests, added by src/CMakeLists.txt with BUILD_TESTING: one GoogleTest executable per
# file, registered per test case and labelled unit. The executables and their discovery files are
# built in <build>/tests/unit, not in the top build directory.
function(add_unit_test name source library)
  add_executable(test_${name} ${source})
  target_link_libraries(test_${name} PRIVATE ${library} GTest::gmock_main)
  target_include_directories(test_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  gtest_discover_tests(test_${name} PROPERTIES LABELS unit DISCOVERY_TIMEOUT 60)
endfunction()
